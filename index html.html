    const locations = {
        entrance: { top: 30, left: 40 },
        dance: { top: 30, left: 800 - 120 },
        food: { top: 190, left: 360 },
        craft: { top: 400 - 65, left: 40 },
        garden: { top: 400 - 65, left: 800 - 120 }
    };

    // Utilizing highly reliable public domain/open instructional audio samples
    const missions = [
        {
            audio: "https://actions.google.com/sounds/v1/ambiences/carnival_midway.ogg",
            instruction: "Welcome! Walk straight ahead from the entrance to reach the delicious Food Stalls.",
            correct: "food",
            options: ["Food Stalls", "Craft Corner"]
        },
        {
            audio: "https://actions.google.com/sounds/v1/doors/heavy_door_open.ogg",
            instruction: "From the Food Stalls, turn right and go past the gardens to find the Traditional Dance Stage.",
            correct: "dance",
            options: ["Dance Stage", "Festival Entrance"]
        },
        {
            audio: "https://actions.google.com/sounds/v1/crowds/crowd_cheering.ogg",
            instruction: "Finally, head over to the Traditional Craft Corner to see the beautiful handmade items.",
            correct: "craft",
            options: ["Traditional Craft Corner", "Festival Garden"]
        }
    ];

    let currentMission = 0;
    let score = 0;
    let lives = 3;

    function startGame() {
        document.getElementById('welcome-screen').classList.remove('active-screen');
        document.getElementById('game-screen').classList.add('active-screen');
        loadMission();
    }

    function loadMission() {
        if (currentMission >= missions.length) {
            endGame(true);
            return;
        }
        document.getElementById('mission-step').innerText = `${currentMission + 1}/${missions.length}`;
        const mission = missions[currentMission];

        const audio = document.getElementById('game-audio');
        const source = document.getElementById('audio-source');
        source.src = mission.audio;
        audio.load(); 

        document.getElementById('instruction-text').innerText = mission.instruction;

        const optionsContainer = document.getElementById('options-container');
        optionsContainer.innerHTML = '';
        mission.options.forEach(opt => {
            const btn = document.createElement('button');
            btn.className = 'option-btn';
            btn.innerText = opt;
            const locKey = opt.toLowerCase().includes('food') ? 'food' : 
                         opt.toLowerCase().includes('dance') ? 'dance' : 
                         opt.toLowerCase().includes('craft') ? 'craft' : 'garden';
            btn.onclick = () => checkAnswer(locKey);
            optionsContainer.appendChild(btn);
        });
    }

    function checkAnswer(selectedLocation) {
        if (selectedLocation === missions[currentMission].correct) {
            score += 100;
            document.getElementById('score').innerText = score;
            moveCharacter(selectedLocation);
            currentMission++;
            setTimeout(loadMission, 1500);
        } else {
            lives--;
            document.getElementById('lives').innerText = "❤".repeat(lives);
            alert("Wrong turn! Try listening again.");
            if (lives <= 0) {
                endGame(false);
            }
        }
    }

    function moveCharacter(locKey) {
        const char = document.getElementById('char');
        const pos = locations[locKey];
        char.style.top = pos.top + 'px';
        char.style.left = pos.left + 'px';
    }

    function endGame(win) {
        document.getElementById('game-screen').classList.remove('active-screen');
        document.getElementById('completion-screen').classList.add('active-screen');
        document.getElementById('final-score').innerText = score;
    }

    function restartGame() {
        currentMission = 0;
        score = 0;
        lives = 3;
        document.getElementById('score').innerText = score;
        document.getElementById('lives').innerText = "❤❤❤";
        document.getElementById('completion-screen').classList.remove('active-screen');
        document.getElementById('welcome-screen').classList.add('active-screen');
    }
